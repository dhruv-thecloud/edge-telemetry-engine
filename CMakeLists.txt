cmake_minimum_required(VERSION 3.20)
project(EdgeTelemetry VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EDGE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(EDGE_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
add_library(edge_options INTERFACE)
if(MSVC)
  target_compile_options(edge_options INTERFACE /W4)
  if(EDGE_WARNINGS_AS_ERRORS)
    target_compile_options(edge_options INTERFACE /WX)
  endif()
else()
  target_compile_options(edge_options INTERFACE -Wall -Wextra -Wpedantic)
  if(EDGE_WARNINGS_AS_ERRORS)
    target_compile_options(edge_options INTERFACE -Werror)
  endif()
endif()
if(EDGE_SANITIZERS)
  if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU" AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(edge_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(edge_options INTERFACE -fsanitize=address,undefined)
  else()
    message(FATAL_ERROR "EDGE_SANITIZERS requires GCC or Clang")
  endif()
endif()
add_library(c_sensors STATIC src/c_sensor.c)
target_include_directories(c_sensors PUBLIC include)
target_link_libraries(c_sensors PRIVATE edge_options)
if(UNIX)
  target_link_libraries(c_sensors PUBLIC m)
endif()
add_library(cpp_aggregator STATIC src/cpp_aggregator.cpp)
target_link_libraries(cpp_aggregator PUBLIC c_sensors PRIVATE edge_options)
add_executable(edge_telemetry src/main.cpp)
target_link_libraries(edge_telemetry PRIVATE cpp_aggregator edge_options)
include(CTest)
if(BUILD_TESTING)
  foreach(test_name IN ITEMS test_c_objects test_cpp_core)
    add_executable(${test_name} tests/${test_name}.cpp)
    target_link_libraries(${test_name} PRIVATE cpp_aggregator edge_options)
    add_test(NAME ${test_name} COMMAND ${test_name})
  endforeach()
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_test(NAME test_protocol COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_protocol.py $<TARGET_FILE:edge_telemetry>)
  set_tests_properties(test_protocol PROPERTIES TIMEOUT 30)
endif()
install(TARGETS edge_telemetry RUNTIME DESTINATION bin)
